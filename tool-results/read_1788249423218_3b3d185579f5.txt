    37→const fmt = (n: number) => n.toLocaleString('ar-EG')
    38→
    39→const relativeTime = (iso: string) => {
    40→  const diff = Date.now() - new Date(iso).getTime()
    41→  const mins = Math.floor(diff / 60000)
    42→  if (mins < 1) return 'الآن'
    43→  if (mins < 60) return `منذ ${fmt(mins)} دقيقة`
    44→  const hrs = Math.floor(mins / 60)
    45→  if (hrs < 24) return `منذ ${fmt(hrs)} ساعة`
    46→  const days = Math.floor(hrs / 24)
    47→  if (days < 30) return `منذ ${fmt(days)} يوم`
    48→  const months = Math.floor(days / 30)
    49→  return `منذ ${fmt(months)} شهر`
    50→}
