    77→const relativeTime = (iso: string) => {
    78→  const diff = Date.now() - new Date(iso).getTime()
    79→  const mins = Math.floor(diff / 60000)
    80→  if (mins < 1) return 'الآن'
    81→  if (mins < 60) return `منذ ${fmt(mins)} دقيقة`
    82→  const hrs = Math.floor(mins / 60)
    83→  if (hrs < 24) return `منذ ${fmt(hrs)} ساعة`
    84→  const days = Math.floor(hrs / 24)
    85→  return `منذ ${fmt(days)} يوم`

   242→function DarkTooltip({
   243→  active,
   244→  payload,
   245→  label,
   246→}: {
   247→  active?: boolean
   248→  payload?: Array<{ value: number; dataKey: string; color: string }>
   249→  label?: string
   250→}) {
   251→  if (!active || !payload?.length) return null
   252→  return (
   253→    <div className="rounded-lg border border-white/10 bg-zinc-900/95 px-4 py-3 shadow-2xl backdrop-blur-md">
   254→      <p className="mb-1 text-xs text-zinc-400">{label}</p>
   255→      {payload.map((p) => (
   256→        <p key={p.dataKey} className="text-sm font-medium" style={{ color: p.color }}>
   257→          {p.dataKey === 'users' ? 'المستخدمون' : 'الإعلانات'}:{' '}
   258→          {fmt(p.value)}
   259→        </p>
   260→      ))}
   261→    </div>
   262→  )
   263→}
