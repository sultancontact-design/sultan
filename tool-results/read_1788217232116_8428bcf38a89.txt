    88→const weeklyData = useMemo(() => {
    89→  const raw = adminChartData.userGrowth
    90→  const weeks: { label: string; users: number }[] = []
    91→  for (let w = 0; w < 8; w++) {
    92→    const start = w * 7
    93→    const slice = raw.slice(start, start + 7)
    94→    const total = slice.reduce((s, d) => s + d.users, 0)
    95→    const first = slice[0]

    98→  }
    99→  return weeks
