    83→/* ------------------------------------------------------------------ */
    84→/*  Weekly aggregation helper (last 8 weeks)                            */
    85→/* ------------------------------------------------------------------ */
    86→function buildWeeklyData() {
    87→  const raw = adminChartData.userGrowth
    88→  const weeks: { label: string; users: number }[] = []
    89→  for (let w = 0; w < 8; w++) {
    90→    const start = w * 7
    91→    const slice = raw.slice(start, start + 7)
    92→    const total = slice.reduce((s, d) => s + d.users, 0)
    93→    const first = slice[0]
    94→    const weekNum = Math.ceil((new Date(first.date).getDate()) / 7)
    95→    weeks.push({ label: `أسبوع ${weekNum}`, users: total })
    96→  }
    97→  return weeks
    98→}
    99→const weeklyData = buildWeeklyData()
