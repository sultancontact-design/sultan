  1049→// ═══════════════════════════════════════════

  1051→/**
  1052→ * Get a translated string by key and language.
  1053→ * Falls back: requested lang → 'ar' → key itself.
  1054→ *
  1055→ * @param key  - The translation key (e.g. 'heroTitle')
  1056→ * @param lang - The target language code ('ar' | 'fr' | 'en' | 'darija')
  1057→ * @returns The translated string, or the key if not found
  1058→ */
  1059→export function t(key: string, lang: SupportedLang = 'ar'): string {
  1060→  // Try the requested language first
  1061→  if (translations[lang]?.[key]) {
  1062→    return translations[lang][key];
  1063→  }
  1064→  // Fallback to Arabic (primary language)
  1065→  if (translations.ar?.[key]) {
  1066→    return translations.ar[key];
  1067→  }
  1068→  // Ultimate fallback: return the key itself so nothing breaks
  1069→  return key;
  1070→}
  1071→
  1072→/**
  1073→ * React hook-compatible translation function factory.
  1074→ * Returns a bound `t` function for a given language.
  1075→ * Usage in components:
  1076→ *   const t = useTranslation('fr');
  1077→ *   <h1>{t('heroTitle')}</h1>
  1078→ */
  1079→export function useTranslation(lang: SupportedLang = 'ar'): (key: string) => string {
  1080→  return (key: string) => t(key, lang);
  1081→}
