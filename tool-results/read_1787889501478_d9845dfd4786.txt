    46→

   209→            <div className="flex-1 min-w-0">
   210→              <div className="flex items-center gap-2 mb-1">
   211→                <span className="text-sm font-bold text-orange-400">قيد التكامل</span>

   218→              </p>

   220→          </div>
   221→        </motion.div>
   222→      )}
   223→
   224→      {/* ═══ Feature Disabled Message ═══ */}
   225→      {!cashoutEnabled && (
   226→        <motion.div {...fadeUp} transition={{ delay: 0.05 }} className="mb-4">

   360→              <span className="text-xs text-muted-foreground">حالة التحقق (KYC)</span>
