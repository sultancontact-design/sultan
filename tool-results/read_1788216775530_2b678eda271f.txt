    66→  const diff = Date.now() - new Date(iso).getTime()

    71→  if (hrs < 24) return `منذ ${fmt(hrs)} ساعة`

   230→function DarkTooltip({
   231→  active,
   232→  payload,
   233→  label,
   234→}: {
   235→  active?: boolean
   236→  payload?: Array<{ value: number; dataKey: string; color: string }>
   237→  label?: string
   238→}) {
   239→  if (!active || !payload?.length) return null
   240→  return (
   241→    <div className="rounded-lg border border-white/10 bg-zinc-900/95 px-4 py-3 shadow-2xl backdrop-blur-md">
   242→      <p className="mb-1 text-xs text-zinc-400">{label}</p>
   243→      {payload.map((p) => (
   244→        <p key={p.dataKey} className="text-sm font-medium" style={{ color: p.color }}>
   245→          {p.dataKey === 'users' ? 'المستخدمون' : 'الإعلانات'}: {fmt(p.value)}
   246→        </p>
   247→      ))}
   248→    </div>
   249→  )
   250→}

   252→/* ------------------------------------------------------------------ */
   253→/*  Health Pill Sub-component                                          */
   254→/* ------------------------------------------------------------------ */
   255→function HealthPill({
   256→  label,
   257→  value,
   258→  status,
   259→}: {
   260→  label: string
   261→  value: string
   262→  status: 'good' | 'warn' | 'bad'
   263→}) {
   264→  const dotColor =
   265→    status === 'good'
   266→      ? 'bg-emerald-500'
   267→      : status === 'warn'
   268→        ? 'bg-amber-500'
   269→        : 'bg-red-500'
   270→  return (
   271→    <div className="rounded-lg bg-white/[0.03] px-3 py-2.5">
   272→      <div className="mb-1 flex items-center gap-1.5">
   273→        <span className={`h-1.5 w-1.5 rounded-full ${dotColor}`} />
   274→        <span className="text-[11px] text-zinc-500">{label}</span>
   275→      </div>
   276→      <p className="text-sm font-semibold text-zinc-200">{value}</p>
   277→    </div>
   278→  )
   279→}
