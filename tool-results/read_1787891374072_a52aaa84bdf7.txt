   129→   158→    power: emptyBalance('SP' as any),

   135→   164→// ─── Store Implementation ────────────────────────────────────────────────────

   137→   166→export const useEconomyStore = create<EconomyState>()((set, get) => ({
   138→   167→  // ─── Wallet ────────────────────────────────────────────────────────────────
   139→   168→  wallet: null,
   140→   169→
   141→   170→  initWallet: (userId: string) => {
   142→   171→    const existing = get().wallet;
   143→   172→    if (existing && existing.userId === userId) return;
   144→   173→    const wallet = createEmptyWallet(userId);
   145→   174→    // Give demo user some coins
   146→   175→    if (userId === 'demo-001') {
   147→   176→      wallet.coins = { ...wallet.coins, balance: 2500, available: 2500, lifetimeReceived: 5000, lifetimeGiven: 2500 };
   148→   177→      wallet.rewards = { ...wallet.rewards, balance: 750, available: 750, lifetimeReceived: 1200 };
   149→   178→      wallet.pendingRewards = { ...wallet.pendingRewards, pending: 200 };
   150→   179→      wallet.power = { ...wallet.power, balance: 1200, available: 1200, lifetimeReceived: 2000 };
   151→   180→    } else if (userId === 'admin-001') {
   152→   181→      wallet.coins = { ...wallet.coins, balance: 99999, available: 99999 };
   153→   182→      wallet.power = { ...wallet.power, balance: 9999, available: 9999 };
   154→   183→    }
   155→   184→    set({ wallet });
   156→   185→  },
   157→   186→
   158→   187→  getWalletBalance: (currency) => {
   159→   188→    const w = get().wallet;
   160→   189→    if (!w) return 0;
   161→   190→    switch (currency) {
   162→   191→      case 'SC': return w.coins.balance;
   163→   192→      case 'SR': return w.rewards.balance;
   164→   193→      case 'SP': return w.power.balance;
   165→   194→      default: return 0;
   166→   195→    }
   167→   196→  },
   168→   197→
   169→   198→  getAvailableBalance: (currency) => {
   170→   199→    const w = get().wallet;
   171→   200→    if (!w) return 0;
   172→   201→    switch (currency) {
   173→   202→      case 'SC': return w.coins.available;
   174→   203→      case 'SR': return w.rewards.available;
   175→   204→      case 'SP': return w.power.available;
   176→   205→      default: return 0;
   177→   206→    }
   178→   207→  },
   179→   208→
   180→   209→  getPendingBalance: (currency) => {
   181→   210→    const w = get().wallet;
   182→   211→    if (!w) return 0;
   183→   212→    return w.pendingRewards.pending;
   184→   213→  },
   185→   214→
   186→   215→  // ─── Ledger ────────────────────────────────────────────────────────────────
   187→   216→  ledger: [],
   188→   217→
   189→   218→  addLedgerEntry: (entry) => {
   190→   219→    const id = generateId();
   191→   220→    const rule = get().getRule('platform_fee_percent');
   192→   221→    const tx: LedgerTransaction = {
   193→   222→      ...entry,
   194→   223→      id,
   195→   224→      createdAt: now(),
   196→   225→      ruleVersion: rule?.version ?? 1,
   197→   226→    };
   198→   227→    set((s) => ({ ledger: [tx, ...s.ledger] }));
   199→   228→    return id;
   200→   229→  },
   201→   230→
   202→   231→  getTransactionHistory: (userId, filters) => {
   203→   232→    let txs = get().ledger.filter((t) => t.userId === userId);
   204→   233→    if (filters?.type) txs = txs.filter((t) => t.type === filters.type);
   205→   234→    if (filters?.currency) txs = txs.filter((t) => t.currency === filters.currency);
   206→   235→    if (filters?.limit) txs = txs.slice(0, filters.limit);
   207→   236→    return txs;
   208→   237→  },
   209→   238→
   210→   239→  // ─── Support ───────────────────────────────────────────────────────────────
   211→   240→  supportEvents: [],
   212→   241→
   213→   242→  executeSupport: (params) => {
   214→   243→    const { supporterId, recipientId, amount, message, targetType, targetId } = params;
   215→   244→    const state = get();
   216→   245→
   217→   246→    // Validation: cannot support self
   218→   247→    if (supporterId === recipientId) {
   219→   248→      return { success: false, transactionId: '', error: 'لا يمكنك دعم نفسك' };
   220→   249→    }
   221→   250→
   222→   251→    // Check feature enabled
   223→   252→    if (!state.isFeatureEnabled('support_enabled')) {
   224→   253→      return { success: false, transactionId: '', error: 'الدعم معطل حالياً' };
   225→   254→    }
   226→   255→
   227→   256→    // Check emergency pause
   228→   257→    if (state.isFeaturePaused('support')) {
   229→   258→      return { success: false, transactionId: '', error: 'الدعم متوقف مؤقتاً' };
   230→   259→    }
   231→   260→
   232→   261→    // Check balance
   233→   262→    const feeRate = state.getRule('platform_fee_percent')?.value ?? 5;
   234→   263→    const fee = Math.round((amount * feeRate) / 100);
   235→   264→    const total = amount + fee;
   236→   265→    const available = state.getAvailableBalance('SC');
   237→   266→
   238→   267→    if (available < total) {
   239→   268→      return { success: false, transactionId: '', error: 'رصيدك غير كافٍ' };
   240→   269→    }
   241→   270→
   242→   271→    // Execute
   243→   272→    const txId = generateId();
   244→   273→    const event: SupportEvent = {
   245→   274→      id: generateId(), supporterId, recipientId, amount, fee,
   246→   275→      recipientReceives: amount,
   247→   276→      currency: 'SC', message, targetType: targetType as any, targetId,
   248→   277→      transactionId: txId, createdAt: now(),
   249→   278→    };
   250→   279→
   251→   280→    // Deduct from supporter
   252→   281→    state.addLedgerEntry({
   253→   282→      userId: supporterId, source: 'wallet', destination: `user:${recipientId}`,
   254→   283→      currency: 'SC', amount: -total, type: 'support_sent', status: 'completed',
   255→   284→      referenceId: targetId, campaignId: null, completedAt: now(),
   256→   285→      createdBy: supporterId, riskStatus: 'low', metadata: { message, targetType, targetId }, auditId: null,
   257→   286→    });
   258→   287→
   259→   288→    // Add to recipient
   260→   289→    state.addLedgerEntry({
   261→   290→      userId: recipientId, source: `user:${supporterId}`, destination: 'wallet',
   262→   291→      currency: 'SC', amount, type: 'support_received', status: 'completed',
   263→   292→      referenceId: targetId, campaignId: null, completedAt: now(),
   264→   293→      createdBy: supporterId, riskStatus: 'low', metadata: { message, targetType, targetId }, auditId: null,
   265→   294→    });
   266→   295→
   267→   296→    // Update wallet balances
   268→   297→    set((s) => {
   269→   298→      if (!s.wallet || s.wallet.userId !== supporterId) return s;
   270→   299→      return {
   271→   300→        wallet: {
   272→   301→          ...s.wallet,
   273→   302→          coins: { ...s.wallet.coins, balance: s.wallet.coins.balance - total, available: s.wallet.coins.available - total, lifetimeGiven: s.wallet.coins.lifetimeGiven + total },
   274→   303→        },
   275→   304→      };
   276→   305→    });
   277→   306→
   278→   307→    // Add support event
   279→   308→    set((s) => ({ supportEvents: [...s.supportEvents, event] }));
   280→   309→
   281→   310→    // Add notification for recipient
   282→   311→    state.addEconomyNotification({
   283→   312→      userId: recipientId, type: 'support_received',
   284→   313→      title: 'دعم جديد',
   285→   314→      body: `تم دعمك بـ ${amount.toLocaleString()} SC`,
   286→   315→      metadata: { supporterId, amount, message, txId },
   287→   316→    });
   288→   317→
   289→   318→    return { success: true, transactionId: txId };
   290→   319→  },
   291→   320→
   292→   321→  getSupporterStats: (userId) => {
   293→   322→    const events = get().supportEvents.filter((e) => e.supporterId === userId);
   294→   323→    return {
   295→   324→      totalGiven: events.reduce((sum, e) => sum + e.amount + e.fee, 0),
   296→   325→      peopleSupported: new Set(events.map((e) => e.recipientId)).size,

   312→   341→      userId, source, destination: 'wallet', currency: 'SR', amount,
   313→   342→      type: 'reward_pending', status: 'pending', referenceId,
   314→   343→      campaignId: null, completedAt: null, createdBy: null,

   331→   360→  cashoutRequests: [],

   333→   362→  getCashoutEligibility: (userId) => {
   334→   363→    const state = get();
   335→   364→    if (!state.isFeatureEnabled('cashout_enabled')) {
   336→   365→      return { eligible: false, reason: 'السحب غير مفعّل حالياً', reviewStatus: 'none' as any, riskStatus: 'low' as any, kycStatus: 'not_required' as any, approvedBy: null, approvedAt: null, expiresAt: null };
   337→   366→    }
   338→   367→    if (state.isFeaturePaused('cashouts')) {
   339→   368→      return { eligible: false, reason: 'السحب متوقف مؤقتاً من قبل الإدارة', reviewStatus: 'none' as any, riskStatus: 'low' as any, kycStatus: 'not_required' as any, approvedBy: null, approvedAt: null, expiresAt: null };
   340→   369→    }
   341→   370→    const risk = state.getRiskScore(userId);
   342→   371→    if (risk.level === 'high' || risk.level === 'critical') {
   343→   372→      return { eligible: false, reason: 'حسابك يحتاج مراجعة أمنية', reviewStatus: 'pending' as any, riskStatus: risk.level, kycStatus: 'not_required' as any, approvedBy: null, approvedAt: null, expiresAt: null };
   344→   373→    }
   345→   374→    return { eligible: true, reason: 'مؤهل', reviewStatus: 'approved' as any, riskStatus: 'low' as any, kycStatus: 'not_required' as any, approvedBy: null, approvedAt: null, expiresAt: null };
   346→   375→  },
   347→   376→
   348→   377→  requestCashout: (userId, amount) => {
   349→   378→    const state = get();
   350→   379→    const eligibility = state.getCashoutEligibility(userId);
   351→   380→    if (!eligibility.eligible) {
   352→   381→      return { success: false, cashoutId: '', error: eligibility.reason };
   353→   382→    }
   354→   383→    const minCashout = (state.getRule('min_cashout_sr')?.value ?? 500) as number;
   355→   384→    if (amount < minCashout) {
   356→   385→      return { success: false, cashoutId: '', error: `الحد الأدنى للسحب ${minCashout.toLocaleString()} SR` };
   357→   386→    }
   358→   387→    const available = state.getAvailableBalance('SR');
   359→   388→    if (available < amount) {
   360→   389→      return { success: false, cashoutId: '', error: 'رصيدك المتاح غير كافٍ' };
   361→   390→    }
   362→   391→
   363→   392→    const id = generateId();
   364→   393→    const cashout: CashoutRequest = {
   365→   394→      id, userId, amount, currency: 'SR', status: 'requested',
   366→   395→      method: 'bank_transfer', methodDetails: {}, eligibility,
   367→   396→      requestedAt: now(), processedAt: null, completedAt: null,
   368→   397→      rejectionReason: null, transactionId: null, auditId: null,
   369→   398→    };
   370→   399→    set((s) => ({
   371→   400→      cashoutRequests: [...s.cashoutRequests, cashout],
   372→   401→      wallet: s.wallet && s.wallet.userId === userId ? {
   373→   402→        ...s.wallet,
   374→   403→        rewards: { ...s.wallet.rewards, available: s.wallet.rewards.available - amount, pending: s.wallet.rewards.pending + amount },
   375→   404→      } : s.wallet,
   376→   405→    }));
   377→   406→    return { success: true, cashoutId: id };
   378→   407→  },
   379→   408→
   380→   409→  updateCashoutStatus: (cashoutId, status, reason) => {
   381→   410→    set((s) => ({
   382→   411→      cashoutRequests: s.cashoutRequests.map((c) =>
   383→   412→        c.id === cashoutId
   384→   413→          ? { ...c, status: status as any, processedAt: now(), rejectionReason: reason ?? c.rejectionReason }
   385→   414→          : c
   386→   415→      ),
   387→   416→    }));
   388→   417→  },
   389→   418→
   390→   419→  // ─── Rules Engine ──────────────────────────────────────────────────────────
   391→   420→  economyRules: [...DEFAULT_ECONOMY_RULES],
   392→   421→
   393→   422→  getRule: (key) => get().economyRules.find((r) => r.key === key),
   394→   423→
   395→   424→  updateRule: (key, newValue, adminId, reason) => {

   399→   428→        const updated: EconomyRule = {
   400→   429→          ...r, value: newValue, previousValue: r.value,
   401→   430→          version: r.version + 1, updatedBy: adminId, updatedAt: now(), changeReason: reason,
   402→   431→        };
   403→   432→        return updated;
